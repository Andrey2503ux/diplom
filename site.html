const steps = document.querySelectorAll('.question-step'); const nextBtn = document.getElementById('next-btn'); const prevBtn = document.getElementById('prev-btn'); let currentStep = 0; function updateNavigation() { steps.forEach((step, idx) => { step.classList.toggle('active', idx === currentStep); }); prevBtn.disabled = currentStep === 0; nextBtn.innerText = currentStep === steps.length - 1 ? "Завершить" : "Далее"; } nextBtn.addEventListener('click', () => { if (currentStep < steps.length - 1) { currentStep++; updateNavigation(); } else { alert("Тест окончен! Проверь свои ответы по ключам."); } }); prevBtn.addEventListener('click', () => { if (currentStep > 0) { currentStep--; updateNavigation(); } }); function checkOption(btn, isCorrect) { // Простая визуальная проверка для кнопок btn.style.backgroundColor = isCorrect ? "#c6f6d5" : "#fed7d7"; btn.style.borderColor = isCorrect ? "#48bb78" : "#f56565"; }
